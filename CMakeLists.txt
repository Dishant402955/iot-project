cmake_minimum_required(VERSION 3.16)
project(wifi_presence_lab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(wifi_core src/signal_sample.cpp src/linux_wifi.cpp src/csv_writer.cpp src/feature_extractor.cpp src/anomaly_detector.cpp src/clock.cpp src/config.cpp src/collector.cpp src/cli.cpp)
target_include_directories(wifi_core PUBLIC include)
target_compile_options(wifi_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(wifi_sense src/main.cpp)
target_link_libraries(wifi_sense PRIVATE wifi_core)
add_executable(wifi_tests tests/test_main.cpp)
target_link_libraries(wifi_tests PRIVATE wifi_core)
enable_testing()
add_test(NAME wifi_tests COMMAND wifi_tests)
